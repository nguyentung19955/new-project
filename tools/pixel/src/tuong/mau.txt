  swap s G
  swap G g
  swap p r
  use chan 0 24
  use tren 0 3
  use mat_nham 14 11
  outline
end

  use vay 0 24
  use tren 0 3
  use mat_nham 14 11

  use canh_ngang 8 26
  outline
end

  use @die.1
  swap n D
  swap N n
  swap G g
  swap s G
  swap p r
  swap Y y
end
