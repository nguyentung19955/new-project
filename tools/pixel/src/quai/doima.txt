anim walk fps=8 loop
frame walk
  use w0
  outline O
end
frame walk
  use w1
  outline O
end
frame walk
  use w2
  outline O
end
frame walk
  use w3
  outline O
end

# đánh: xếp cánh lao lên → vỗ mạnh xuống, há nanh cắn → về
anim attack fps=8 once
frame attack
  use a0
  outline O
end
frame attack
  use a1
  outline O
end
frame attack
  use @walk.0
end

anim hurt fps=8 once
frame hurt
  use @walk.1
  shift -1 0
  swap P p
  swap p E
  swap R e
end

# chết: cánh rủ, mắt tắt → rơi xuống nằm → tối màu
anim die fps=5 once
frame die
  use d0
  outline O
end
frame die
  use d1
  rot 180
  shift 0 2
  outline O
end
frame die
  use @die.1
  swap e E
  swap p P
end
